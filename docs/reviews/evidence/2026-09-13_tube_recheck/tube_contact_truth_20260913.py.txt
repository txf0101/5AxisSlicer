from pathlib import Path
import json
import sys

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'src'))
from OCP.BRepClass3d import BRepClass3d_SolidClassifier
from OCP.TopAbs import TopAbs_IN
from OCP.gp import gp_Pnt
from five_axis_slicer.manufacturing.setup import ManufacturingSetup, TubeOperationDefinition
from five_axis_slicer.step_loader import load_step
from five_axis_slicer.tube_controller import TubeSetupController
from five_axis_slicer.tube_generation_context import make_tube_generation_context
from five_axis_slicer.postprocessing.indexed_tube import _generate_geometry

out = ROOT / 'docs/reviews/evidence/2026-09-13_tube_recheck'
inputs = json.loads((out / 'pipe2_own_ac_spatial/inputs.json').read_text(encoding='utf-8'))
model = load_step(inputs['source'])
operation = TubeOperationDefinition.from_json(inputs['operation'])
controller = TubeSetupController(model, setup=ManufacturingSetup.from_json(inputs['setup']), operations=(operation,))
context = make_tube_generation_context(controller, operation)
feature, plan, path = _generate_geometry(context.model_in_build, operation, None)
point = path.points[1]
shank_center = tuple(p - 2 * a for p, a in zip(point.position, point.nozzle_axis))
substrate = context.model_in_build.shapes['body_001']
tip_inside = BRepClass3d_SolidClassifier(substrate, gp_Pnt(*point.position), 1e-7).State() == TopAbs_IN
shank_inside = BRepClass3d_SolidClassifier(substrate, gp_Pnt(*shank_center), 1e-7).State() == TopAbs_IN
report = {
    'method': 'Independent OCCT solid point containment, not collision AABB',
    'first_deposition_point': point.to_json(), 'nozzle_shank_center_mm': shank_center,
    'tip_inside_actual_substrate': tip_inside,
    'shank_center_inside_actual_substrate': shank_inside,
    'input_sha256': model.source_hash,
    'interpretation': 'Both tested centers are outside the substrate. This center-only test cannot rule out sphere-envelope intersection. The recorded nozzle axis is radial rather than the required fixed Indexed build direction; this run predates the orientation repair.',
}
(out / 'pipe2_contact_truth.json').write_text(json.dumps(report, indent=2), encoding='utf-8')
print(json.dumps(report, indent=2))
