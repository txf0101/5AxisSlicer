def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str), encoding='utf8')

def summary(result):
    manifest=result.manifest.to_json();manifest.pop('toolpath',None)
    return {'exportable': result.exportable, 'manifest': manifest,
            'readback': result.readback.to_json(), 'validation': result.validation.to_json(),
            'points': len(result.toolpath.points), 'events': len(result.toolpath.events)}

def plot_planar(result, out, title):
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    from matplotlib.collections import LineCollection
    fig, ax = plt.subplots(figsize=(10, 8), layout='constrained')
    layer = next(x for x in result.layers if x.regions)
    for region in layer.regions:
        for loop in (region.outer, *region.holes):
            ax.plot([p[0] for p in loop], [p[1] for p in loop], color='black', lw=1.3)
    dep, travel = [], []
    for a,b in zip(result.toolpath.points, result.toolpath.points[1:]):
        if b.layer_id != layer.layer_id: continue
        (dep if b.point_type == 'deposition' else travel).append([a.position[:2], b.position[:2]])
    ax.add_collection(LineCollection(dep, colors='#0077b6', linewidths=.9, label='Deposition centreline'))
    ax.add_collection(LineCollection(travel, colors='#e76f51', linewidths=.6, linestyles='dashed', label='Travel'))
    ax.autoscale(); ax.set_aspect('equal'); ax.set_xlabel('Build X / mm'); ax.set_ylabel('Build Y / mm')
    ax.set_title(f'{title}\nZ={layer.z_mm:g} mm | black = CAD section'); ax.legend(loc='best')
    fig.savefig(out / 'path.png', dpi=150); plt.close(fig)

def mutate_readback(result, out, marker='P02'):
    def e_double(text):
        return re.sub(r'(?<=\s)E(-?\d+(?:\.\d+)?)', lambda m:f'E{2*float(m[1]):.6f}', text)
    lines=result.gcode.splitlines(); removed=[]; skip=False
    for line in lines:
        if ' EVENT ' in line and (' retract' in line or ' prime' in line): skip=True; continue
        if skip: skip=False; continue
        removed.append(line)
    variants={'extrusion_x2':e_double(result.gcode),
              'relative_axes':result.gcode.replace('G90 ; absolute axes','G91 ; relative axes'),
              'relative_extrusion':result.gcode.replace('M82 ; absolute extrusion','M83 ; relative extrusion'),
              'events_removed':'\n'.join(removed)+'\n',
              'extra_unmarked_move':result.gcode.replace('M400', 'G1 X999 Y999 Z999 E999\nM400')}
    records={}
    for name,code in variants.items():
        report=readback_indexed_gcode(code,result.toolpath,result.trajectory,_machine(),marker_tag=marker)
        records[name]=report.to_json()
        (out / f'INVALID_{name}.txt').write_text(code,encoding='utf8')
    records['original']=result.readback.to_json()
    records['extruder_zero_reset_present']=bool(re.search(r'^G92\s+.*E', result.gcode,re.M))
    return records

def planar(case, model_path, body, first, last, kind='planar_zigzag', default_feed=False):
    folder=OUT/case; folder.mkdir(parents=True,exist_ok=True)
    model=load_step(ROOT/model_path)
    controller=PlanarController(model,setup=_setup(model,body))
    op=controller.create_operation(kind,operation_id=case)
    parameters=PlanarProcessParameters(first_layer_z_mm=first,last_layer_z_mm=last,
        layer_height_mm=.6,bead_width_mm=.6,line_spacing_mm=.6,
        feedrate_mm_min=1200 if default_feed else 100,travel_feedrate_mm_min=1800 if default_feed else 100)
    op=controller.configure_operation(operation_id=op.operation_id,body_id=body,parameters=parameters)
    result=controller.generate_operation(op.operation_id)
    data=summary(result); data.update(model=model_path,source_sha256=model.source_hash,body=body,
        parameters=parameters.to_json(),layers=len(result.layers))
    write(folder/'result.json',result.to_json()); write(folder/'toolpath.json',result.toolpath.to_json())
    if result.exportable: controller.export_operation_product(op.operation_id,str(folder/'product'))
    plot_planar(result,folder,case)
    return data

def analytic():
    import cadquery as cq
    folder=OUT/'analytic'; folder.mkdir(parents=True,exist_ok=True)
    path=folder/'rectangle_8x6x1.step'
    cq.exporters.export(cq.Workplane('XY').box(8,6,1,centered=False),str(path))
    model=load_step(path); body=model.bodies[0].body_id
    controller=PlanarController(model,setup=_setup(model,body))
    op=controller.create_operation('planar_zigzag',operation_id='analytic-rectangle')
    op=controller.configure_operation(operation_id=op.operation_id,body_id=body,
        parameters=PlanarProcessParameters(first_layer_z_mm=.5,last_layer_z_mm=.5,layer_height_mm=.5,
        bead_width_mm=.6,feedrate_mm_min=100,travel_feedrate_mm_min=100))
    result=controller.generate_operation(op.operation_id)
    if result.exportable: controller.export_operation_product(op.operation_id,str(folder/'product'))
    data=summary(result)
    data['mutations']=mutate_readback(result,folder)
    data['expected_xy_bounds_mm']=[0,0,8,6]
    segments=[(a,b) for a,b in zip(result.toolpath.points,result.toolpath.points[1:]) if b.point_type=='deposition']
    # Independent analytic bounding box of circular bead sweeps.
    data['bead_sweep_xy_bounds_mm']=[min(min(a.position[k],b.position[k])-b.bead_width_mm/2 for a,b in segments) for k in (0,1)]+[max(max(a.position[k],b.position[k])+b.bead_width_mm/2 for a,b in segments) for k in (0,1)]
    data['target_volume_single_layer_mm3']=8*6*.5
    data['emitted_material_volume_mm3']=sum(p.material_volume_mm3 for p in result.toolpath.points)
    plot_planar(result,folder,'Analytic 8 x 6 mm rectangle')
    import matplotlib.pyplot as plt
    fig,ax=plt.subplots(figsize=(9,6),layout='constrained')
    from matplotlib.patches import Rectangle
    ax.add_patch(Rectangle((-.3,-.3),8.6,6.6,facecolor='#e76f51',alpha=.28,label='Outer envelope of boundary beads'))
    ax.add_patch(Rectangle((0,0),8,6,fill=False,edgecolor='black',lw=2,label='CAD boundary 8 x 6 mm'))
    ax.set(xlim=(-1,9),ylim=(-1,7),xlabel='X / mm',ylabel='Y / mm',title='Zigzag boundary: bead width 0.6 mm\nMeasured XY overshoot: 0.3 mm per side; internal outside metric: 0')
    ax.set_aspect('equal');ax.legend();fig.savefig(folder/'bead_envelope.png',dpi=150);plt.close(fig)
    return data

def tube(case, controller_mode=False):

        data=summary(result);data['parameters']=op.parameters.to_json()
