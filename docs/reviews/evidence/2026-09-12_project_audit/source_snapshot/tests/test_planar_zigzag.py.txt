from pathlib import Path
import sys

import pytest

sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

from five_axis_slicer.algorithms.planar import (
    PlanarRegion,
    PlanarSliceLayer,
    PlanarZigzagError,
    ZigzagParameters,
    generate_zigzag_toolpath,
)
from five_axis_slicer.validation import measure_planar_toolpath


def _loop(*points):
    return tuple((*points, points[0]))


def test_zigzag_clips_hole_and_separates_travel_from_deposition():
    region = PlanarRegion(
        "region-1",
        _loop((0, 0, 1), (10, 0, 1), (10, 10, 1), (0, 10, 1)),
        (_loop((4, 4, 1), (6, 4, 1), (6, 6, 1), (4, 6, 1)),),
    )
    path = generate_zigzag_toolpath(
        "op", (PlanarSliceLayer("layer-1", 1, (region,)),), ZigzagParameters(1, 0.5, 0.2)
    )
    deposition = [point for point in path.points if point.point_type == "deposition"]
    assert deposition and all(point.material_volume_mm3 > 0 for point in deposition)
    assert all(
        not (4 < point.position[0] < 6 and 4 < point.position[1] < 6) for point in deposition
    )
    assert all(
        point.material_volume_mm3 == 0 for point in path.points if point.point_type != "deposition"
    )


def test_empty_or_invalid_input_has_localised_diagnostic():
    with pytest.raises(PlanarZigzagError, match="planar.zigzag_layers_missing"):
        generate_zigzag_toolpath("op", (), ZigzagParameters(1, 0.5, 0.2))
    with pytest.raises(PlanarZigzagError, match="planar.zigzag_parameter_invalid"):
        ZigzagParameters(0, 0.5, 0.2)


def test_perimeters_close_each_loop_and_every_transition_is_explicit():
    region = PlanarRegion(
        "concave",
        _loop((0, 0, 0), (8, 0, 0), (8, 3, 0), (3, 3, 0), (3, 8, 0), (0, 8, 0)),
        (_loop((1, 1, 0), (2, 1, 0), (2, 2, 0), (1, 2, 0)),),
    )
    other = PlanarRegion("island", _loop((20, 0, 0), (22, 0, 0), (22, 2, 0), (20, 2, 0)))
    path = generate_zigzag_toolpath(
        "op", (PlanarSliceLayer("layer", 0, (other, region)),), ZigzagParameters(1, 0.5, 0.2)
    )
    perimeter = [point for point in path.points if point.extrusion_role == "skin"]
    assert len(perimeter) == 14  # 6 + 4 + 4 closed-loop edges
    event_types = [event.event_type for event in path.events]
    assert event_types.count("retract") == event_types.count("prime") - 1
    assert all(
        point.material_volume_mm3 == 0 for point in path.points if point.point_type != "deposition"
    )
    measurement = measure_planar_toolpath(path, (PlanarSliceLayer("layer", 0, (other, region)),))
    assert measurement.deposition_outside_max_mm == pytest.approx(0)
    assert measurement.volume_difference_mm3 == pytest.approx(0)
    assert measurement.covered_area_mm2 > 0
